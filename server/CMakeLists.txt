cmake_minimum_required(VERSION 3.16)
project(hermes-touch-voice VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Qml Quick Multimedia Network)
add_executable(hermes-touch-voice app/main.cpp app/Voice.cpp app/Voice.h app/resources.qrc)
target_link_libraries(hermes-touch-voice PRIVATE Qt5::Core Qt5::Gui Qt5::Qml Qt5::Quick Qt5::Multimedia Qt5::Network)
install(TARGETS hermes-touch-voice RUNTIME DESTINATION .)
install(FILES manifest.json app/voice.desktop app/voice.apparmor app/icon.svg DESTINATION .)
